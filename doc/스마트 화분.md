# 스마트 화분 토이 프로젝트

## 초기 문의 내용
현재 첨부된 예제 코드만을 활용해서 "스마트 화분" 토이 프로젝트를 진행하려고 합니다.
개발 가능한 수준과 샘플 코드 예시 작성해 주세요.

## 개발 가능한 기능

첨부된 코드와 데이터시트를 분석한 결과, 다음과 같은 스마트 화분 기능을 구현할 수 있습니다:

1. **환경 모니터링**
   - DHT11 센서를 활용한 온습도 측정
   - XADC를 이용한 토양 수분 측정
   - HC-SR04 초음파 센서를 이용한 물 수위 측정

2. **사용자 인터페이스**
   - LCD 디스플레이를 통한 식물 상태 정보 표시
   - 키패드를 통한 설정값 조정

3. **자동화 시스템**
   - 토양 수분에 따른 자동 급수 제어
   - 온도/습도에 따른 환경 경고 시스템

## 시스템 구조

```
[센서부]       [제어부]        [출력부]
DHT11 센서  →              → LCD 디스플레이
토양수분센서 → 메인 컨트롤러 → LED 상태표시
초음파 센서  →              → 급수 모터 제어
키패드      →
```

### 2. 메인 컨트롤러 모듈

```verilog
module smart_pot_controller(
    input wire clk,
    input wire reset,
    input wire [7:0] humidity,
    input wire [7:0] temperature,
    input wire [15:0] distance,
    input wire [7:0] soil_moisture,
    input wire [7:0] keypad_value,
    output reg water_pump,
    output reg [7:0] led,
    output wire water_level_warning,
    output wire soil_dry_warning
);
    
    // 사용자 설정 임계값 (기본값)
    reg [7:0] soil_moisture_threshold = 30;  // 토양 수분 임계값 (30% 미만이면 건조)
    reg [15:0] water_level_threshold = 100;  // 물 수위 임계값 (100mm 미만이면 경고)
    
    // 상태 정의
    localparam IDLE = 2'b00;
    localparam WATERING = 2'b01;
    localparam WAIT_AFTER_WATERING = 2'b10;
    
    reg [1:0] state = IDLE;
    reg [31:0] wait_counter = 0;
    
    // 경고 신호 생성
    assign soil_dry_warning = (soil_moisture < soil_moisture_threshold);
    assign water_level_warning = (distance > water_level_threshold);
    
    // 키패드 입력 처리 (임계값 설정)
    always @(posedge clk) begin
        if (reset) begin
            soil_moisture_threshold <= 30;
            water_level_threshold <= 100;
        end else begin
            case (keypad_value)
                8'h1: soil_moisture_threshold <= 20;  // 키 1: 토양 수분 임계값 20%
                8'h2: soil_moisture_threshold <= 30;  // 키 2: 토양 수분 임계값 30%
                8'h3: soil_moisture_threshold <= 40;  // 키 3: 토양 수분 임계값 40%
                8'h4: water_level_threshold <= 80;    // 키 4: 물 수위 임계값 80mm
                8'h5: water_level_threshold <= 100;   // 키 5: 물 수위 임계값 100mm
                8'h6: water_level_threshold <= 120;   // 키 6: 물 수위 임계값 120mm
                default: ; // 다른 키는 무시
            endcase
        end
    end
    
    // 메인 제어 로직 (상태 기계)
    always @(posedge clk) begin
        if (reset) begin
            state <= IDLE;
            water_pump <= 0;
            wait_counter <= 0;
            led <= 8'h00;
        end else begin
            case (state)
                IDLE: begin
                    // LED 상태 표시
                    led[0] <= soil_dry_warning;        // 토양 건조 경고
                    led[1] <= water_level_warning;     // 물 수위 경고
                    led[7:2] <= 6'b000000;
                    
                    // 급수 시작 조건: 토양이 건조하고 물 수위가 충분함
                    if (soil_dry_warning && !water_level_warning) begin
                        state <= WATERING;
                        water_pump <= 1;
                        led[2] <= 1; // 급수 중 표시
                    end
                end
                
                WATERING: begin
                    // 5초 동안 급수
                    if (wait_counter >= 5 * 100000000) begin // 5초 (100MHz 클럭 기준)
                        state <= WAIT_AFTER_WATERING;
                        water_pump <= 0;
                        wait_counter <= 0;
                        led[2] <= 0;
                    end else begin
                        wait_counter <= wait_counter + 1;
                    end
                end
                
                WAIT_AFTER_WATERING: begin
                    // 30초 대기 (토양 수분이 측정값에 반영될 시간)
                    if (wait_counter >= 30 * 100000000) begin // 30초
                        state <= IDLE;
                        wait_counter <= 0;
                    end else begin
                        wait_counter <= wait_counter + 1;
                        // LED 깜박임 - 대기 중 표시
                        if (wait_counter % 50000000 < 25000000) // 0.5초 주기
                            led[3] <= 1;
                        else
                            led[3] <= 0;
                    end
                end
                
                default: state <= IDLE;
            endcase
        end
    end

endmodule
```

### 3. DHT11 센서 컨트롤러

```verilog
module dht11_controller(
    input wire clk,
    input wire reset,
    inout wire dht11_data,  // 양방향 데이터 핀
    output reg [7:0] humidity,
    output reg [7:0] temperature
);
    
    // DHT11 통신 상태
    localparam IDLE = 4'd0;
    localparam START_SIGNAL = 4'd1;
    localparam RESPONSE_WAIT = 4'd2;
    localparam RECEIVE_DATA = 4'd3;
    localparam PROCESS_DATA = 4'd4;
    localparam WAIT_NEXT = 4'd5;
    
    reg [3:0] state = IDLE;
    reg dht11_out = 1;
    reg dht11_in_en = 0;  // 데이터 방향 제어 (0: 출력, 1: 입력)
    reg [39:0] received_data = 0;
    reg [5:0] bit_count = 0;
    reg [31:0] timer = 0;
    
    // DHT11 데이터 핀 제어 (양방향)
    assign dht11_data = dht11_in_en ? 1'bz : dht11_out;
    wire dht11_in = dht11_data;
    
    always @(posedge clk) begin
        if (reset) begin
            state <= IDLE;
            dht11_out <= 1;
            dht11_in_en <= 0;
            humidity <= 0;
            temperature <= 0;
            received_data <= 0;
            bit_count <= 0;
            timer <= 0;
        end else begin
            case (state)
                IDLE: begin
                    // 2초마다 센서 읽기 시작
                    if (timer >= 200000000) begin // 2초 (100MHz 기준)
                        timer <= 0;
                        state <= START_SIGNAL;
                        dht11_out <= 0;  // 시작 신호: Low
                        dht11_in_en <= 0; // 출력 모드
                    end else begin
                        timer <= timer + 1;
                        dht11_out <= 1;
                        dht11_in_en <= 0;
                    end
                end
                
                START_SIGNAL: begin
                    // 시작 신호 18ms 이상 유지
                    if (timer >= 1800000) begin // 18ms
                        timer <= 0;
                        dht11_out <= 1;
                        state <= RESPONSE_WAIT;
                        dht11_in_en <= 1; // 입력 모드로 전환
                    end else begin
                        timer <= timer + 1;
                    end
                end
                
                RESPONSE_WAIT: begin
                    // DHT11 응답 대기
                    if (timer < 100000) begin // 대기 타임아웃: 1ms
                        timer <= timer + 1;
                        // DHT11이 Low 신호를 보내고 다시 High로 전환하는 것을 감지
                        if (!dht11_in && timer > 20000) begin // Low 감지
                            state <= RECEIVE_DATA;
                            timer <= 0;
                            bit_count <= 0;
                            received_data <= 0;
                        end
                    end else begin
                        // 응답 없음, 다시 시도
                        state <= IDLE;
                        timer <= 0;
                    end
                end
                
                RECEIVE_DATA: begin
                    // 40비트 데이터 수신
                    if (bit_count < 40) begin
                        if (dht11_in) begin // 상승 엣지 감지
                            timer <= timer + 1;
                        end else begin // 하강 엣지 감지
                            if (timer > 4000) begin // 약 40us 이상은 '1'로 간주
                                received_data <= (received_data << 1) | 1'b1;
                            end else begin
                                received_data <= (received_data << 1);
                            end
                            bit_count <= bit_count + 1;
                            timer <= 0;
                        end
                    end else begin
                        state <= PROCESS_DATA;
                    end
                end
                
                PROCESS_DATA: begin
                    // 데이터 검증 및 처리
                    // 체크섬: DHT11 마지막 8비트는 앞의 4바이트 합의 하위 8비트
                    if ((received_data[39:32] + received_data[31:24] + 
                         received_data[23:16] + received_data[15:8]) & 8'hFF == received_data[7:0]) begin
                        humidity <= received_data[39:32];      // 습도 정수부
                        temperature <= received_data[23:16];   // 온도 정수부
                    end
                    state <= WAIT_NEXT;
                    timer <= 0;
                end
                
                WAIT_NEXT: begin
                    // 다음 읽기 전 대기
                    if (timer >= 100000000) begin // 1초 대기
                        state <= IDLE;
                        timer <= 0;
                    end else begin
                        timer <= timer + 1;
                    end
                end
                
                default: state <= IDLE;
            endcase
        end
    end

endmodule
```

   - LCD 인터페이스 완성 (HD44780 프로토콜 구현)
